// Classe BombermanGame before refacto 1
import { tileSize, tilesPerRow, cols, rows, zoomRate } from "../utils/vars.js";
export class BombermanGame extends Phaser.Scene {
    constructor() {
        super({ key: "BombermanGame" });

        // Affectation des locales
        this.canDiagonalMove = false;
        this.movementLocked = false;

        // Affectation des constantes
        this.tileSize = tileSize;
        this.tilesPerRow = tilesPerRow;
        this.cols = cols;
        this.rows = rows;
        this.zoomRate = zoomRate;

        // Gestion des bombes (Affectation locale)
        this.bombCooldown = false; // Empêche le spam de bombes
        this.bombDuration = 2000; // Temps avant l'explosion [ms]
        this.bombSize = 1; // Rayon initial de la bombe
    }

    preload() {
        // Chargement de la spritesheet
        this.load.spritesheet("sprites", "../src/js/Bomber/src/img/bomber_asset.png", {
            frameWidth: this.tileSize,
            frameHeight: this.tileSize
        });
    }
    
    create() {
        this.bombKey = this.input.keyboard.addKey(Phaser.Input.Keyboard.KeyCodes.SPACE); // Touche bombe (espace)
        this.bombs = this.physics.add.group(); // Liste des bombes posées

        this.createMap(); // Création de la carte en premier pour placer correctement le joueur
        this.createPlayer(); // Placement du joueur après la map

        // Appliquer les collisions
        this.physics.add.collider(this.player, this.walls);
        this.physics.add.collider(this.player, this.obstacles);
        this.physics.add.collider(this.bombs, this.walls);
        this.physics.add.collider(this.bombs, this.obstacles);

        this.createAnimations();

        this.cursors = this.input.keyboard.createCursorKeys();
    }

    createAnimations(){
        // Création des animations de déplacement :
        this.anims.create({ // Vers le haut
            key: "walk-up",
            frames: this.anims.generateFrameNumbers(
                "sprites", { start: 6 * this.tilesPerRow, end: 6 * this.tilesPerRow + 3 }
            ),
            frameRate: 10,
            repeat: -1
        });

        this.anims.create({ // Vers la droite
            key: "walk-right",
            frames: this.anims.generateFrameNumbers(
                "sprites", { start: 6 * this.tilesPerRow + 4, end: 6 * this.tilesPerRow + 7 }
            ),
            frameRate: 10,
            repeat: -1
        });

        this.anims.create({ // Vers le bas
            key: "walk-down",
            frames: this.anims.generateFrameNumbers(
                "sprites", { start: 6 * this.tilesPerRow + 8, end: 6 * this.tilesPerRow + 11 }
            ),
            frameRate: 10,
            repeat: -1
        });

        this.anims.create({ // Vers la gauche
            key: "walk-left",
            frames: this.anims.generateFrameNumbers(
                "sprites", { start: 6 * this.tilesPerRow + 12, end: 6 * this.tilesPerRow + 15 }
            ),
            frameRate: 10,
            repeat: -1
        });

        // Animation de la bombe (6 frames)
        this.anims.create({
            key: "bomb-timer",
            frames: this.anims.generateFrameNumbers("sprites", {
                start: 7 * this.tilesPerRow + 4,
                end: 7 * this.tilesPerRow + 9
            }),
            frameRate: 3,
            repeat: -1
        });

        // Animation des explosions
        this.anims.create({
            key: "explosion-center",
            frames: [{ key: "sprites", frame: 8 * this.tilesPerRow + 1 }],
            frameRate: 10,
            repeat: 0
        });

        this.anims.create({
            key: "explosion-vertical",
            frames: [{ key: "sprites", frame: 8 * this.tilesPerRow }],
            frameRate: 10,
            repeat: 0
        });

        this.anims.create({
            key: "explosion-horizontal",
            frames: [{ key: "sprites", frame: 9 * this.tilesPerRow + 2 }],
            frameRate: 10,
            repeat: 0
        });

        this.anims.create({
            key: "explosion-top",
            frames: [{ key: "sprites", frame: 7 * this.tilesPerRow }],
            frameRate: 10,
            repeat: 0
        });

        this.anims.create({
            key: "explosion-bottom",
            frames: [{ key: "sprites", frame: 9 * this.tilesPerRow }],
            frameRate: 10,
            repeat: 0
        });

        this.anims.create({
            key: "explosion-left",
            frames: [{ key: "sprites", frame: 9 * this.tilesPerRow + 1 }],
            frameRate: 10,
            repeat: 0
        });

        this.anims.create({
            key: "explosion-right",
            frames: [{ key: "sprites", frame: 9 * this.tilesPerRow + 3 }],
            frameRate: 10,
            repeat: 0
        });
    }

    createMap() {
        this.cameras.main.setBackgroundColor("#4b5320"); // Fond herbe
        
        this.walls = this.physics.add.staticGroup(); // Murs solides
        this.obstacles = this.physics.add.staticGroup(); // Obstacles destructibles
    
        let tileWallBottom = 14 * this.tilesPerRow;
        let tileWallLeft = 14 * this.tilesPerRow + 1;
        let tileWallTop = 14 * this.tilesPerRow + 2;
        let tileWallRight = 14 * this.tilesPerRow + 3;
        let tileWallCorner = 10 * this.tilesPerRow + 4;
    
        let spawnSafeZone = [{ x: 1, y: 1 }, { x: 2, y: 1 }, { x: 1, y: 2 }, { x: 2, y: 2 }];
    
        for (let i = 0; i < this.cols; i++) {
            for (let j = 0; j < this.rows; j++) {
                const x = i * this.tileSize + this.tileSize / 2; // Positionnement centré
                const y = j * this.tileSize + this.tileSize / 2;

                if(spawnSafeZone.some(p => p.x === i && p.y === j)) continue;
    
                if (i === 0 && j === 0 ||
                    i === this.cols - 1 && j === 0 ||
                    i === 0 && j === this.rows - 1 ||
                    i === this.cols - 1 && j === this.rows - 1){
                    this.createWall(x, y, tileWallCorner);
                } else if (i === 0) {
                    this.createWall(x, y, tileWallLeft);
                } else if(j === 0) {
                    this.createWall(x, y, tileWallTop);
                } else if(i === this.cols - 1) {
                    this.createWall(x, y, tileWallRight);
                } else if(j === this.rows - 1) {
                    this.createWall(x, y, tileWallBottom);
                } else if (i % 2 === 0 && j % 2 === 0) {
                    this.createWall(x, y, 28);
                } else if (Math.random() > 0.7) {
                    this.createObstacle(x, y, 12); // Bloc destructible
                }
            }
        }
    }

    createWall(x, y, frame){
        const wall = this.walls.create(x, y, 'sprites', frame)
            .setOrigin(0.5)
            .setSize(this.tileSize, this.tileSize, true);
        wall.body.immovable = true;
        return wall;
    }

    createObstacle(x, y, frame){
        const obstacle = this.obstacles.create(x, y, 'sprites', frame)
            .setOrigin(0.5)
            .setSize(this.tileSize, this.tileSize, true);
        obstacle.body.immovable = true;
        return obstacle;
    }

    createPlayer() {
        // Position initiale CENTRÉE dans la tuile (1,1)
        const startX = (1 * this.tileSize) + this.tileSize / 2; // 24
        const startY = (1 * this.tileSize) + this.tileSize / 2; // 24

        this.player = this.physics.add.sprite(startX, startY, "sprites", 6 * this.tilesPerRow + 8)
            .setOrigin(0.5)
            .setSize(12, 12) // Hitbox réduite de 4px
            .setOffset(2, 2); // Décalage pour centrer la hitbox
        
        this.cameras.main.startFollow(this.player, true, 0.1, 0.1); // Lissage X/Y
        this.cameras.main.setBounds(0, 0, this.rows * this.tileSize, this.rows * this.tileSize); // Taille totale de la carte
        this.cameras.main.setZoom(this.zoomRate); // Zoom optionnel pour mieux voir
    }

    handlePlayerMovement() {
        const speed = 100;
        this.player.setVelocity(0);
        const moved = this.handleDirectionInput(speed);

        if(!moved) {        
            this.snapToGridX();
            this.snapToGridY();
            this.player.anims.stop();
        }
    }

    handleDirectionInput(speed) {
        let moved = false;

        // INVERSER L'ORDRE DES CHECK ET SIMPLIFIER
        if (this.cursors.left.isDown) {
            this.tryMoveHorizontal(-speed, 'walk-left');
            return true;
        }
        if (this.cursors.right.isDown) {
            this.tryMoveHorizontal(speed, 'walk-right');
            return true;
        }
        if (this.cursors.up.isDown) {
            this.tryMoveVertical(-speed, 'walk-up');
            return true;
        }
        if (this.cursors.down.isDown) {
            this.tryMoveVertical(speed, 'walk-down');
            return true;
        }
        return moved;
    }

    isAlignedHorizontally() {
        return Math.round(this.player.x) % this.tileSize === this.tileSize / 2;
    }
    
    isAlignedVertically() {
        return Math.round(this.player.y) % this.tileSize === this.tileSize / 2;
    }    
    
    tryMoveHorizontal(speed, anim) {
        if (this.isAlignedVertically()) {
            this.snapToGridY(); // Force l'alignement avant déplacement
            this.player.setVelocityX(speed);
            this.player.anims.play(anim, true);
        }
    }

    tryMoveVertical(speed, anim) {
        if (this.isAlignedHorizontally()) {
            //console.log("Moving vertically with speed:", speed);
            this.player.setVelocityY(speed);
            this.player.anims.play(anim, true);
            this.snapToGridX(); // Alignement horizontal forcé
        }
    }

    snapToGridX() {
        const tileIndex = Math.floor(this.player.x / this.tileSize);
        const snapX = tileIndex * this.tileSize + this.tileSize / 2;
        this.player.x = snapX;
    }

    snapToGridY() {
        const tileIndex = Math.floor(this.player.y / this.tileSize);
        const snapY = tileIndex * this.tileSize + this.tileSize / 2;
        this.player.y = snapY;
    }

    placeBomb() {
        if (this.bombCooldown) return;

        const tileX = Math.floor(this.player.x / this.tileSize) * this.tileSize + this.tileSize / 2;
        const tileY = Math.floor(this.player.y / this.tileSize) * this.tileSize + this.tileSize / 2;

        let bomb = this.bombs.create(tileX, tileY, "sprites", 7 * this.tilesPerRow + 4)
            .setOrigin(0.5)
            .setImmovable(true);
        
        bomb.play("bomb-timer"); // Animation de la bombe

        this.bombCooldown = true;
        this.time.delayedCall(this.bombDuration, () => this.explodeBomb(bomb), [], this);
        this.time.delayedCall(500, () => this.bombCooldown = false, [], this);
    }

    explodeBomb(bomb) {
        if(!bomb) return;
        
        // Suppression de la bombe
        bomb.destroy();

        // Créer l'explosion
        this.createExplosion(bomb.x, bomb.y, 'explosion-center');

        // Propagation de l'explosion en fonction de la puissance
        this.propagateExplosion(bomb.x, bomb.y);
    }

    createExplosion(x, y, animationKey) {
        let explosion = this.add.sprite(x, y, "sprites").setOrigin(0.5);
        explosion.play(animationKey);
        this.time.delayedCall(500, () => explosion.destroy(), [], this);
    }

    propagateExplosion(x, y) {
        let directions = [
            { dx: 1, dy: 0, startKey: "explosion-left", midKey: "explosion-horizontal", endKey: "explosion-right" },
            { dx: -1, dy: 0, startKey: "explosion-right", midKey: "explosion-horizontal", endKey: "explosion-left" },
            { dx: 0, dy: 1, startKey: "explosion-top", midKey: "explosion-vertical", endKey: "explosion-bottom" },
            { dx: 0, dy: -1, startKey: "explosion-bottom", midKey: "explosion-vertical", endKey: "explosion-top" }
        ];

        for (let dir of directions) {
            for (let i = 1; i <= this.bombSize; i++) {
                let newX = x + dir.dx * i * this.tileSize;
                let newY = y + dir.dy * i * this.tileSize;

                // Vérifier si un mur indestructible bloque l'explosion
                if (this.walls.getChildren().some(wall => wall.x === newX && wall.y === newY)) break;

                // Vérifier si un obstacle destructible doit être détruit
                let obstacle = this.obstacles.getChildren().find(obs => obs.x === newX && obs.y === newY);
                if (obstacle) {
                    obstacle.destroy(); // Supprimer l'obstacle
                    this.createExplosion(newX, newY, dir.endKey);
                    break; // L'explosion s'arrête ici
                }
                // Créer une explosion sur la case
                let explosionKey = (i === this.bombSize) ? dir.endKey : dir.midKey;
                this.createExplosion(newX, newY, explosionKey);
            }
        }
    }
    
    update() {
        this.handlePlayerMovement();

        if(Phaser.Input.Keyboard.JustDown(this.bombKey)){
            this.placeBomb();
        }
    }
}
